  <script>
    var SURAH_LIST = [];
    var currentSurah = null;
    var currentVerse = null;
    var isHidden = false;
    var totalDihafalCount = 0;

    // Media Recorder Variables
    var mediaRecorder = null;
    var audioChunks = [];
    var isRecording = false;
    var recTimerInterval = null;
    var recSeconds = 0;
    var mediaStream = null;

    document.addEventListener("DOMContentLoaded", function() {
      muatDaftarSurah();
      muatStatistik();
    });

    function showView(viewId) {
      stopRecording();
      var syekhPlayer = document.getElementById('syekh-audio-player');
      if (syekhPlayer) syekhPlayer.pause();

      document.querySelectorAll('.step-view').forEach(el => el.classList.remove('active'));
      document.getElementById(viewId).classList.add('active');
    }

    function muatStatistik() {
      totalDihafalCount = parseInt(localStorage.getItem('total_hafalan_count') || 0);
      document.getElementById('stat-total').innerText = totalDihafalCount;
    }

    function muatDaftarSurah() {
      fetch('https://equran.id/api/v2/surat')
        .then(res => res.json())
        .then(resData => {
          if (resData && resData.data) {
            SURAH_LIST = resData.data;
            renderSurahList(SURAH_LIST);
          }
        })
        .catch(err => {
          document.getElementById('surah-list').innerHTML = '<div style="color:red; text-align:center;">Gagal memuat surat. Cek koneksi internet.</div>';
        });
    }

    function renderSurahList(list) {
      var container = document.getElementById('surah-list');
      container.innerHTML = '';
      list.forEach(function(item) {
        var div = document.createElement('div');
        div.className = 'surah-item';
        div.innerHTML = '<div><strong>' + item.nomor + '. ' + item.namaLatin + ' (' + item.nama + ')</strong><br><small style="color:#7F8C8D">' + item.jumlahAyat + ' Ayat • ' + item.tempatTurun + '</small></div>➔';
        div.onclick = function() { selectSurah(item); };
        container.appendChild(div);
      });
    }

    function filterSurah() {
      var query = document.getElementById('search-input').value.toLowerCase();
      var filtered = SURAH_LIST.filter(function(s) {
        return s.namaLatin.toLowerCase().includes(query) || s.nomor.toString().includes(query);
      });
      renderSurahList(filtered);
    }

    function selectSurah(surah) {
      currentSurah = surah;
      document.getElementById('selected-surah-title').innerText = surah.nomor + '. ' + surah.namaLatin + ' (' + surah.jumlahAyat + ' Ayat)';
      
      var grid = document.getElementById('grid-ayat-container');
      grid.innerHTML = '';
      for (var i = 1; i <= surah.jumlahAyat; i++) {
        (function(num) {
          var btn = document.createElement('button');
          btn.className = 'btn-ayat';
          btn.innerText = num;
          btn.onclick = function() { selectVerse(num); };
          grid.appendChild(btn);
        })(i);
      }
      showView('view-ayat-select');
    }

    function selectVerse(verseNo) {
      currentVerse = verseNo;

      document.getElementById('practice-title').innerText = currentSurah.namaLatin + " : Ayat " + verseNo;
      document.getElementById('arabic-text').innerText = "⏳ Memuat ayat...";
      document.getElementById('translation-text').innerText = "";

      // Reset Players
      var userAudio = document.getElementById('user-audio-player');
      userAudio.style.display = 'none'; userAudio.src = '';

      var syekhPlayer = document.getElementById('syekh-audio-player');
      syekhPlayer.style.display = 'none'; syekhPlayer.src = '';

      // Reset UI state
      isHidden = false;
      document.getElementById('arabic-text').style.filter = 'none';
      document.getElementById('btn-toggle-hide').innerText = '🙈 Sembunyikan Teks Ayat';
      document.getElementById('syekh-status').innerText = '';
      document.getElementById('rec-error').style.display = 'none';

      showView('view-practice');

      fetch('https://equran.id/api/v2/surat/' + currentSurah.nomor)
        .then(res => res.json())
        .then(resData => {
          if (resData && resData.data && resData.data.ayat) {
            var verseData = resData.data.ayat[verseNo - 1];
            document.getElementById('arabic-text').innerText = verseData.teksArab;
            document.getElementById('translation-text').innerText = '"' + verseData.teksIndonesia + '"';
          }
        })
        .catch(err => {
          document.getElementById('arabic-text').innerText = "Gagal memuat teks ayat.";
        });
    }

    function toggleBlurAyat() {
      var arText = document.getElementById('arabic-text');
      var btn = document.getElementById('btn-toggle-hide');
      isHidden = !isHidden;
      arText.style.filter = isHidden ? 'blur(8px)' : 'none';
      btn.innerText = isHidden ? '👁 Tampilkan Teks Ayat' : '🙈 Sembunyikan Teks Ayat';
    }

    // FUNGSI REKAM AUDIO (MENGGUNAKAN MEDIARECORDER MURNI)
    function toggleRecord() {
      if (!isRecording) {
        startRecording();
      } else {
        stopRecording();
      }
    }

    function startRecording() {
      var errEl = document.getElementById('rec-error');
      errEl.style.display = 'none';

      if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
        errEl.innerHTML = "⚠️ Browser ini tidak mendukung fitur perekaman suara.";
        errEl.style.display = 'block';
        return;
      }

      var constraints = { 
        audio: { echoCancellation: true, noiseSuppression: true } 
      };

      navigator.mediaDevices.getUserMedia(constraints)
        .then(function(stream) {
          mediaStream = stream;
          audioChunks = [];

          var options = {};
          if (typeof MediaRecorder.isTypeSupported === 'function') {
            if (MediaRecorder.isTypeSupported('audio/webm;codecs=opus')) {
              options = { mimeType: 'audio/webm;codecs=opus' };
            } else if (MediaRecorder.isTypeSupported('audio/webm')) {
              options = { mimeType: 'audio/webm' };
            } else if (MediaRecorder.isTypeSupported('audio/mp4')) {
              options = { mimeType: 'audio/mp4' };
            }
          }

          try {
            mediaRecorder = new MediaRecorder(stream, options);
          } catch (e) {
            mediaRecorder = new MediaRecorder(stream);
          }

          mediaRecorder.ondataavailable = function(e) {
            if (e.data && e.data.size > 0) {
              audioChunks.push(e.data);
            }
          };

          mediaRecorder.onstop = function() {
            var mimeType = mediaRecorder.mimeType || 'audio/wav';
            var audioBlob = new Blob(audioChunks, { type: mimeType });
            var audioUrl = URL.createObjectURL(audioBlob);
            var player = document.getElementById('user-audio-player');
            player.src = audioUrl;
            player.style.display = 'block';
          };

          mediaRecorder.start(100);
          isRecording = true;

          document.getElementById('btn-record').innerText = "⏹ Selesai Rekam";
          document.getElementById('record-timer').style.display = 'block';

          recSeconds = 0;
          recTimerInterval = setInterval(function() {
            recSeconds++;
            var m = String(Math.floor(recSeconds / 60)).padStart(2, '0');
            var s = String(recSeconds % 60).padStart(2, '0');
            document.getElementById('timer-val').innerText = m + ":" + s;
          }, 1000);

        })
        .catch(function(err) {
          console.error("Recording error:", err);
          var pesan = "";
          if (err.name === "NotAllowedError" || err.name === "PermissionDeniedError") {
            pesan = "⚠️ <strong>Izin Mikrofon Ditolak.</strong><br>Silakan izinkan akses mikrofon di browser Anda.";
          } else if (err.name === "NotFoundError" || err.name === "DevicesNotFoundError") {
            pesan = "⚠️ <strong>Mikrofon tidak ditemukan.</strong>";
          } else {
            pesan = "⚠️ <strong>Gagal merekam:</strong> " + err.message;
          }
          errEl.innerHTML = pesan;
          errEl.style.display = 'block';
        });
    }

    function stopRecording() {
      if (mediaRecorder && isRecording) {
        mediaRecorder.stop();
        if (mediaStream) {
          mediaStream.getTracks().forEach(track => track.stop());
        }
        isRecording = false;
        clearInterval(recTimerInterval);

        document.getElementById('btn-record').innerText = "🎙️ Rekam Ulang";
        document.getElementById('record-timer').style.display = 'none';
      }
    }

    function playSyekhAudio() {
      if(!currentSurah || !currentVerse) return;

      var statusEl = document.getElementById('syekh-status');
      var player = document.getElementById('syekh-audio-player');
      
      statusEl.innerText = "⏳ Memuat audio Syekh...";
      
      var apiUrl = "https://api.alquran.cloud/v1/ayah/" + currentSurah.nomor + ":" + currentVerse + "/ar.alafasy";

      fetch(apiUrl)
        .then(res => res.json())
        .then(data => {
          if(data && data.data && data.data.audio) {
            player.src = data.data.audio;
            player.style.display = 'block';
            player.play();
            statusEl.innerText = "🔊 Memutar Audio Syekh (Mishary Rashid)";
          } else {
            statusEl.innerText = "❌ Audio tidak ditemukan.";
          }
        })
        .catch(err => {
          statusEl.innerText = "❌ Gagal memuat audio. Pastikan terhubung internet.";
        });

      player.onended = function() {
        statusEl.innerText = "✓ Selesai diputar";
      };
    }

    function tandaiSelesai() {
      totalDihafalCount++;
      localStorage.setItem('total_hafalan_count', totalDihafalCount);
      muatStatistik();
      alert("Alhamdulillah! Hafalan Ayat " + currentVerse + " dari Surat " + currentSurah.namaLatin + " telah disimpan.");
      showView('view-home');
    }
